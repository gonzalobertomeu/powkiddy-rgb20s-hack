# Demo del H9 con el proyecto demo (GROOVEBOX_DEMO=1): pantalla Mezcla parada y sonando, volumen,
# paneo y envíos de las pistas, la ganancia master, las filas de efectos, los sticks en vivo y la
# Mezcla con la fuente de 12×24.
# Uso sin placa de sonido ni escritorio (motor al ritmo real, sin salida):
#   SDL_VIDEODRIVER=offscreen RGB20S_SIM=1 GROOVEBOX_AUDIO=null GROOVEBOX_DEMO=1 \
#     GROOVEBOX_DATA=/tmp/gb-datos GROOVEBOX_SCRIPT=.../demo-h9.txt groovebox

wait 300

# Grilla → Mezcla (Select + ↓)
press select
tap down

wait 200

# Start: la canción suena y se mueven los medidores y las voces
tap start
wait 1500
shot h9-mezcla-sonando.bmp
